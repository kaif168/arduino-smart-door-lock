ESP32 Arduino C++ Firmware Sketch (esp32_smart_door_lock.ino) 

/*
 * Engineering Project: Arduino-Based Smart Door Lock Firmware
 * Language: C++ (ESP32 Arduino Framework)
 */

#include <WiFi.h>
#include <WebServer.h>
#include <ESP32Servo.h>
#include <MFRC522.h>
#include <Keypad.h>
#include <LiquidCrystal_I2C.h>

const char* ssid = "YOUR_WIFI_SSID";
const char* password = "YOUR_WIFI_PASSWORD";

WebServer server(80);
Servo doorServo;

void setup() {
  Serial.begin(115200);
  doorServo.attach(13);
  doorServo.write(90); // Locked State

  WiFi.begin(ssid, password);
  while (WiFi.status() != WL_CONNECTED) { delay(500); }

  server.on("/api/status", HTTP_GET, [](){
    server.send(200, "application/json", "{\"status\":\"OK\"}");
  });
  server.begin();
}

void loop() {
  server.handleClient();
}
